cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
include(compilerconfig)
include(defaults)
include(helpers)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin-main.cpp src/panel.cpp src/panel.hpp)
target_compile_features(${CMAKE_PROJECT_NAME} PRIVATE cxx_std_17)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network shell32 ole32)
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
add_executable(casterdesk-panel-checks src/panel-checks.cpp src/panel.cpp src/panel.hpp)
target_compile_features(casterdesk-panel-checks PRIVATE cxx_std_17)
target_link_libraries(casterdesk-panel-checks PRIVATE Qt6::Core Qt6::Widgets Qt6::Network shell32 ole32)
enable_testing()
add_test(NAME panel COMMAND casterdesk-panel-checks)
set_tests_properties(panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
